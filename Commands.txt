All commands below are for macOS (zsh/bash). The gateway runs in Docker, so every
`openclaw ...` command is run inside the container from the openclaw-gateway-docker folder:

cd "../openclaw-gateway-docker"
docker compose exec openclaw-gateway node dist/index.js <openclaw-args>

(Don't use `docker compose run --rm openclaw-cli ...` while the gateway is up: it fails
with GatewayLockNamespaceError.)

# What the gateway uses today: groq/openai/gpt-oss-120b, with ollama/gpt-oss:20b
# (Ollama on this Mac, reached at host.docker.internal:11434) as the only fallback.
# (Llama 3.3 was retired from Groq. Both are gpt-oss reasoning models; `npm run whatsapp:sync`
# sets agents.defaults.thinkingDefault = "low" to keep their hidden reasoning short.)
# Groq free tier for it: 8,000 tokens per minute, 200,000 per day; one reply ~2,900 tokens.
# On a 429 the gateway retries Groq up to 10 times, then answers with Ollama. It goes to Ollama
# at once when Groq asks to wait > 60 s or the daily limit is used up, and while Groq cools down.

# Show the live setting:
docker compose exec openclaw-gateway node dist/index.js config get agents.defaults.model

# List Groq / Ollama models the gateway knows:
docker compose exec openclaw-gateway node dist/index.js models list --provider groq
docker compose exec openclaw-gateway node dist/index.js models list --provider ollama

# Ollama on the Mac (must be running for the fallback):
curl localhost:11434/api/version
ollama list                 # gpt-oss:20b must be listed
ollama pull gpt-oss:20b     # if it isn't
ollama ps                   # is the model loaded right now?

# Restore exactly this setting after experiments (no restart needed):

docker compose exec -T openclaw-gateway node dist/index.js config patch --stdin < ../openclaw-express/openclaw-model-pin.json5

# Prerequisites:
#   Gateway container running and healthy:  docker compose ps   (in openclaw-gateway-docker)
#   Express running (in openclaw-express):  npm run dev
#   API key loaded into this shell (from openclaw-express; every /api call needs it):
#     export API_KEY="$(grep '^API_KEY=' .env | cut -d= -f2-)"
#   The number is in AUTOPILOT_NUMBERS:     npm run match:add -- +98XXXXXXXXXX  (then restart Express)

# Preview the next message (nothing is sent):
curl -s -X POST http://localhost:8080/api/autopilot/message \
  -H "Authorization: Bearer $API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"to":"+98XXXXXXXXXX","context":"Likes hiking","send":false}'

# Send it (opener for a new number, next message for an existing conversation):
curl -s -X POST http://localhost:8080/api/autopilot/message \
  -H "Authorization: Bearer $API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"to":"+98XXXXXXXXXX","context":"Likes hiking"}'

# Read the stored conversation (%2B is "+"):
curl -s -H "Authorization: Bearer $API_KEY" "http://localhost:8080/api/autopilot/conversations/%2B98XXXXXXXXXX"

# Replace +98XXXXXXXXXX with the real WhatsApp number (E.164 format: +countrycode + number, no spaces).
# Test number: +989337203961

body='{"question":"say hi"}'
uri="http://localhost:8080/api/chat/ask"
{
  echo "=== REQUEST ==="
  echo "URI: $uri"
  echo "BODY: $body"
  echo "=== RESPONSE ==="
} > request-log.txt
response=$(curl -s -X POST "$uri" -H "Authorization: Bearer $API_KEY" -H "Content-Type: application/json" -d "$body")
echo "$response" >> request-log.txt
echo "Response: $response"
echo "Saved to request-log.txt"
